    private static void Throws<T>(Action action, string name) where T : Exception
    {
        tests++;
        try { action(); }
        catch (T) { return; }
        throw new Exception(name + ": expected " + typeof(T).Name);
    }

    private static void RoutingTests()
    {
        var index = new SessionRoutingIndex<object>();
        object ownerA = new object();
        object ownerB = new object();
        Equal(true, index.TryGetOrCreate(ownerA, out uint sessionA, out bool created, out string reason), "routing creates A");
        Equal(1u, sessionA, "routing IDs start at one");
        Equal(true, created, "routing A is created");
        Equal("", reason, "routing creation reason empty");
        Equal(true, index.TryGetOrCreate(ownerB, out uint sessionB, out created, out reason), "routing creates B");
        Equal(2u, sessionB, "routing IDs are monotonic");
        Equal(true, index.TryGetOrCreate(ownerA, out uint duplicateA, out created, out reason), "routing duplicate owner returns");
        Equal(sessionA, duplicateA, "routing duplicate owner ID");
        Equal(false, created, "routing duplicate owner is not created");
        Equal("", reason, "routing duplicate reason empty");

        var equalOwners = new SessionRoutingIndex<EqualOwner>();
        var equalA = new EqualOwner(7);
        var equalB = new EqualOwner(7);
        equalOwners.TryGetOrCreate(equalA, out uint equalSessionA, out _, out _);
        equalOwners.TryGetOrCreate(equalB, out uint equalSessionB, out _, out _);
        Equal(1u, equalSessionA, "routing reference owner first ID");
        Equal(2u, equalSessionB, "routing equal values have distinct reference owners");

        Equal(true, index.TryRemoveSession(sessionA, out object removedA), "routing removes A");
        Equal(ownerA, removedA, "routing remove returns owner");
        Equal(true, index.TryGetOrCreate(new object(), out uint sessionC, out _, out _), "routing creates after remove");
        Equal(3u, sessionC, "routing removed ID is not reused");
        Equal(false, index.TryRemoveSession(sessionA, out _), "routing unknown remove rejects");

        var maxIndex = new SessionRoutingIndex<object>(uint.MaxValue);
        object maxOwner = new object();
        Equal(true, maxIndex.TryGetOrCreate(maxOwner, out uint maxSession, out created, out reason), "routing max ID allocates once");
        Equal(uint.MaxValue, maxSession, "routing max ID value");
        Equal(false, maxIndex.TryGetOrCreate(new object(), out uint exhaustedId, out created, out reason), "routing exhaustion rejects new owner");
        Equal(0u, exhaustedId, "routing exhaustion has no ID");
        Equal("session_id_exhausted", reason, "routing exhaustion reason");
        Equal(true, maxIndex.TryGetOrCreate(maxOwner, out uint existingMax, out created, out reason), "routing exhaustion preserves owner lookup");
        Equal(uint.MaxValue, existingMax, "routing exhaustion existing ID");
        Equal(false, created, "routing exhaustion existing is not created");
        Throws<ArgumentOutOfRangeException>(() => new SessionRoutingIndex<object>(0), "routing zero first ID rejects");

        Equal(true, index.TryAttachDrone(sessionB, 20, out reason), "routing attaches B drone");
        Equal(true, index.TryAttachDrone(sessionC, 10, out reason), "routing attaches C drone");
        Equal(true, index.TryResolveDrone(20, out uint resolvedSession), "routing resolves attached drone");
        Equal(sessionB, resolvedSession, "routing resolve has source session");
        Equal(false, index.TryResolveDrone(999, out _), "routing unknown drone does not resolve");
        Equal(false, index.TryAttachDrone(sessionB, 0, out reason), "routing zero drone rejects");
        Equal("invalid_drone_pid", reason, "routing zero drone reason");
        Equal(false, index.TryAttachDrone(999, 30, out reason), "routing unknown attach session rejects");
        Equal("unknown_session", reason, "routing unknown attach session reason");
        Equal(false, index.TryAttachDrone(sessionB, 20, out reason), "routing same-session duplicate rejects");
        Equal("drone_already_attached", reason, "routing same-session duplicate reason");
        Equal(false, index.TryAttachDrone(sessionC, 20, out reason), "routing cross-session duplicate rejects");
        Equal("drone_already_attached", reason, "routing cross-session duplicate reason");
        EqualArray(new uint[] { 20 }, index.SnapshotDronePids(sessionB), "routing duplicate leaves B index unchanged");
        EqualArray(new uint[] { 10 }, index.SnapshotDronePids(sessionC), "routing duplicate leaves C index unchanged");
        Equal(false, index.TryDetachDrone(sessionC, 20, out reason), "routing wrong-source detach rejects");
        Equal("wrong_session", reason, "routing wrong-source detach reason");
        Equal(false, index.TryDetachDrone(sessionB, 999, out reason), "routing unknown detach rejects");
        Equal("unknown_drone", reason, "routing unknown detach reason");
        Equal(true, index.TryDetachDrone(sessionB, 20, out reason), "routing detach source succeeds");
        Equal(true, index.TryAttachDrone(sessionC, 20, out reason), "routing detached drone attaches target");
        Equal(true, index.TryResolveDrone(20, out resolvedSession), "routing transferred drone resolves");
        Equal(sessionC, resolvedSession, "routing transferred drone has only target owner");

        Equal(true, index.TryAttachDrone(sessionB, 40, out _), "routing attaches B cleanup drone");
        Equal(true, index.TryAttachDrone(sessionC, 30, out _), "routing attaches C cleanup drone");
        Equal(true, index.TryRemoveSession(sessionC, out _), "routing removes C with drones");
        Equal(false, index.TryResolveDrone(10, out _), "routing remove clears C first drone");
        Equal(false, index.TryResolveDrone(20, out _), "routing remove clears C transferred drone");
        Equal(false, index.TryResolveDrone(30, out _), "routing remove clears C second drone");
        Equal(true, index.TryResolveDrone(40, out resolvedSession), "routing remove preserves B drone");
        Equal(sessionB, resolvedSession, "routing preserved B drone source");
        EqualArray(new uint[] { 2 }, index.SnapshotSessionIds(), "routing session snapshot sorted");
        EqualArray(new uint[] { 40 }, index.SnapshotDronePids(sessionB), "routing drone snapshot sorted");
        EqualArray(Array.Empty<uint>(), index.SnapshotDronePids(999), "routing unknown drone snapshot empty");

        var sortedIndex = new SessionRoutingIndex<object>();
        object sortedOwnerA = new object();
        object sortedOwnerB = new object();
        sortedIndex.TryGetOrCreate(sortedOwnerA, out uint sortedA, out _, out _);
        sortedIndex.TryGetOrCreate(sortedOwnerB, out uint sortedB, out _, out _);
        sortedIndex.TryAttachDrone(sortedA, 9, out _);
        sortedIndex.TryAttachDrone(sortedA, 2, out _);
        sortedIndex.TryAttachDrone(sortedA, 5, out _);
        EqualArray(new uint[] { 1, 2 }, sortedIndex.SnapshotSessionIds(), "routing session snapshot ascending");
        EqualArray(new uint[] { 2, 5, 9 }, sortedIndex.SnapshotDronePids(sortedA), "routing drone snapshot ascending");
        Equal(true, sortedIndex.TryGetByOwner(sortedOwnerB, out uint lookupSession), "routing owner lookup");
        Equal(sortedB, lookupSession, "routing owner lookup session");
        Equal(true, sortedIndex.TryGetOwner(sortedA, out object lookupOwner), "routing session reverse lookup");
        Equal(sortedOwnerA, lookupOwner, "routing session reverse lookup owner");
        Equal(false, sortedIndex.TryGetByOwner(new object(), out _), "routing unknown owner lookup");
        Equal(false, sortedIndex.TryGetOwner(999, out _), "routing unknown session lookup");
        sortedIndex.Clear();
        Equal(0, sortedIndex.SessionCount, "routing clear removes sessions");
        Equal(false, sortedIndex.TryResolveDrone(2, out _), "routing clear removes drone routes");
        EqualArray(Array.Empty<uint>(), sortedIndex.SnapshotSessionIds(), "routing clear session snapshot empty");
        Equal(true, sortedIndex.TryGetOrCreate(new object(), out uint afterClear, out _, out _), "routing creates after clear");
        Equal(3u, afterClear, "routing clear does not reset allocator");
    }

    private static void EqualArray(uint[] expected, uint[] actual, string name)
    {
        tests++;
        if (expected.Length != actual.Length)
            throw new Exception(name + ": expected length " + expected.Length + ", actual " + actual.Length);
        for (int i = 0; i < expected.Length; i++)
            if (expected[i] != actual[i])
                throw new Exception(name + ": expected " + expected[i] + " at " + i + ", actual " + actual[i]);
    }

    private sealed class EqualOwner
    {
        private readonly int value;

        internal EqualOwner(int value) => this.value = value;

        public override bool Equals(object? obj) => obj is EqualOwner other && other.value == value;

        public override int GetHashCode() => value;
    }
